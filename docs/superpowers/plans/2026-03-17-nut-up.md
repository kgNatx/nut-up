# nut-up Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build nut-up, a bare-metal wrapper around NUT (Network UPS Tools) that automates server setup via CLI wizard, provides a modern dark-themed web dashboard on port 3494 with real-time monitoring, and enables one-command client enrollment.

**Architecture:** Python FastAPI backend talks to NUT's upsd daemon via async TCP on localhost:3493. CLI wizard auto-detects USB UPS via nut-scanner and generates all 5 NUT config files. Web UI is vanilla HTML/CSS/JS served as static files. Client enrollment uses time-limited reusable keys — clients run a single curl command that returns a setup script. State (keys, clients, settings) lives in a JSON file. Everything installs self-contained under /opt/nut-up/ with a Python venv.

**Tech Stack:** Python 3.11+, FastAPI, uvicorn, asyncio TCP sockets, vanilla HTML/CSS/JS, WebSocket, systemd

**Security note:** The web UI renders dynamic data (UPS variable values, hostnames, etc.) via innerHTML. All dynamic values MUST be passed through an `escapeHtml()` helper before interpolation to prevent XSS, even though data sources are trusted (local upsd, admin-controlled state file). Defense in depth.

```python
from __future__ import annotations

import json
import tempfile
from pathlib import Path
from typing import Any, Callable

DEFAULT_STATE = {
    "keys": {},
    "clients": {},
    "server": {},
}


class StateManager:
    def __init__(self, path: Path):
        self.path = Path(path)

    def load(self) -> dict[str, Any]:
        if not self.path.exists():
            self.save(DEFAULT_STATE.copy())
            return DEFAULT_STATE.copy()
        return json.loads(self.path.read_text())

    def save(self, data: dict[str, Any]) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        # Atomic write: write to temp file in same dir, then rename
        fd, tmp = tempfile.mkstemp(dir=self.path.parent, suffix=".tmp")
        try:
            with open(fd, "w") as f:
                json.dump(data, f, indent=2)
            Path(tmp).replace(self.path)
        except BaseException:
            Path(tmp).unlink(missing_ok=True)
            raise

    def update(self, fn: Callable[[dict], dict]) -> dict[str, Any]:
        data = self.load()
        data = fn(data)
        self.save(data)
        return data
```

### Task 3: NUT upsd Protocol Client

**Files:**
- Create: `src/nut_up/upsd_client.py`
- Create: `tests/conftest.py`

This is the core — an async TCP client that speaks the NUT text protocol to upsd on port 3493. It must handle: listing UPS devices, reading variables, setting variables, running instant commands, listing connected clients, and authentication.

**NUT protocol summary:**
- Text-based, line-oriented over TCP
- LIST commands: `LIST UPS` -> `BEGIN LIST UPS` / `UPS name "desc"` / `END LIST UPS`
- GET commands: `GET VAR ups var` -> `VAR ups var "value"`
- SET commands: `SET VAR ups var "value"` -> `OK` (requires auth)
- INSTCMD: `INSTCMD ups cmd` -> `OK` (requires auth)
- Auth: `USERNAME user` -> `OK`, `PASSWORD pass` -> `OK`
- Errors: `ERR CODE-NAME`
- Values are double-quoted, backslash-escaped

```python
import asyncio
import re

import pytest

from nut_up.state import StateManager


@pytest.fixture
def state_file(tmp_path):
    return tmp_path / "state.json"


@pytest.fixture

class MockUpsd:
    """A minimal mock NUT upsd server for testing."""

    def __init__(self):
        self.host = "127.0.0.1"
        self.port = 0  # OS-assigned
        self._server = None
        self._ups_data: dict[str, dict[str, str]] = {
            "myups": {
                "ups.status": "OL",
                "battery.charge": "100",
                "battery.runtime": "3600",
                "ups.load": "25",
                "input.voltage": "120.0",
                "output.voltage": "120.0",
                "ups.temperature": "30.5",
                "battery.voltage": "27.4",
                "ups.mfr": "APC",
                "ups.model": "Back-UPS BX750MI",
                "ups.serial": "9B2416A22014",
                "device.type": "ups",
                "ups.description": "Server Room UPS",
            },
        }
        self._rw_vars: dict[str, set[str]] = {
            "myups": {"battery.charge.low", "ups.delay.shutdown"},
        }
        self._commands: dict[str, set[str]] = {
            "myups": {"test.battery.start", "test.battery.stop", "beeper.toggle"},
        }
        self._users = {"admin": "testpass"}

    async def start(self):
        self._server = await asyncio.start_server(
            self._handle_client, self.host, self.port
        )
        addr = self._server.sockets[0].getsockname()
        self.port = addr[1]

    async def stop(self):
        if self._server:
            self._server.close()
            await self._server.wait_closed()

    async def _handle_client(
        self, reader: asyncio.StreamReader, writer: asyncio.StreamWriter
    ):
        authed = False
        username = None
        try:
            while True:
                line = await reader.readline()
                if not line:
                    break
                cmd = line.decode().strip()
                response = self._process(cmd, authed, username)
                if cmd.startswith("USERNAME "):
                    username = cmd.split(" ", 1)[1]
                if cmd.startswith("PASSWORD ") and username in self._users:
                    if cmd.split(" ", 1)[1] == self._users[username]:
                        authed = True
                writer.write((response + "\n").encode())
                await writer.drain()
        except (ConnectionResetError, asyncio.IncompleteReadError):
            pass
        finally:
            writer.close()

    def _process(self, cmd: str, authed: bool, username: str | None) -> str:
        parts = cmd.split()
        if not parts:
            return "ERR UNKNOWN-COMMAND"

        if parts[0] == "VER":
            return "mock-upsd 0.1 (nut-up test)"

        if parts[0] == "LIST":
            return self._handle_list(parts)

        if parts[0] == "GET":
            return self._handle_get(parts)

        if parts[0] == "USERNAME":
            name = parts[1] if len(parts) > 1 else ""
            if name in self._users:
                return "OK"
            return "ERR INVALID-USERNAME"

        if parts[0] == "PASSWORD":
            pw = parts[1] if len(parts) > 1 else ""
            if username and self._users.get(username) == pw:
                return "OK"
            return "ERR INVALID-PASSWORD"

        if parts[0] == "SET":
            if not authed:
                return "ERR ACCESS-DENIED"
            if len(parts) >= 5 and parts[1] == "VAR":
                ups_name, var_name = parts[2], parts[3]
                value = " ".join(parts[4:]).strip('"')
                if ups_name in self._ups_data:
                    self._ups_data[ups_name][var_name] = value
                    return "OK"
            return "ERR INVALID-ARGUMENT"

        if parts[0] == "INSTCMD":
            if not authed:
                return "ERR ACCESS-DENIED"
            if len(parts) >= 3:
                ups_name, cmd_name = parts[1], parts[2]
                if ups_name in self._commands and cmd_name in self._commands[ups_name]:
                    return "OK"
                return "ERR CMD-NOT-SUPPORTED"
            return "ERR INVALID-ARGUMENT"

        if parts[0] == "LOGOUT":
            return "OK Goodbye"

        return "ERR UNKNOWN-COMMAND"

    def _handle_list(self, parts: list[str]) -> str:
        if len(parts) < 2:
            return "ERR INVALID-ARGUMENT"

        if parts[1] == "UPS":
            lines = ["BEGIN LIST UPS"]
            for name, data in self._ups_data.items():
                desc = data.get("ups.description", name)
                lines.append(f'UPS {name} "{desc}"')
            lines.append("END LIST UPS")
            return "\n".join(lines)

        if parts[1] == "VAR" and len(parts) >= 3:
            ups_name = parts[2]
            if ups_name not in self._ups_data:
                return "ERR UNKNOWN-UPS"
            lines = [f"BEGIN LIST VAR {ups_name}"]
            for var, val in self._ups_data[ups_name].items():
                lines.append(f'VAR {ups_name} {var} "{val}"')
            lines.append(f"END LIST VAR {ups_name}")
            return "\n".join(lines)

        if parts[1] == "RW" and len(parts) >= 3:
            ups_name = parts[2]
            if ups_name not in self._ups_data:
                return "ERR UNKNOWN-UPS"
            lines = [f"BEGIN LIST RW {ups_name}"]
            for var in self._rw_vars.get(ups_name, set()):
                val = self._ups_data[ups_name].get(var, "")
                lines.append(f'RW {ups_name} {var} "{val}"')
            lines.append(f"END LIST RW {ups_name}")
            return "\n".join(lines)

        if parts[1] == "CMD" and len(parts) >= 3:
            ups_name = parts[2]
            if ups_name not in self._commands:
                return "ERR UNKNOWN-UPS"
            lines = [f"BEGIN LIST CMD {ups_name}"]
            for cmd in self._commands[ups_name]:
                lines.append(f"CMD {ups_name} {cmd}")
            lines.append(f"END LIST CMD {ups_name}")
            return "\n".join(lines)

        if parts[1] == "CLIENT" and len(parts) >= 3:
            ups_name = parts[2]
            lines = [f"BEGIN LIST CLIENT {ups_name}"]
            lines.append(f"CLIENT {ups_name} 127.0.0.1")
            lines.append(f"END LIST CLIENT {ups_name}")
            return "\n".join(lines)

        return "ERR INVALID-ARGUMENT"

    def _handle_get(self, parts: list[str]) -> str:
        if len(parts) < 3:
            return "ERR INVALID-ARGUMENT"

        if parts[1] == "VAR" and len(parts) >= 4:
            ups_name, var_name = parts[2], parts[3]
            if ups_name not in self._ups_data:
                return "ERR UNKNOWN-UPS"
            val = self._ups_data[ups_name].get(var_name)
            if val is None:
                return "ERR VAR-NOT-SUPPORTED"
            return f'VAR {ups_name} {var_name} "{val}"'

        if parts[1] == "UPSDESC" and len(parts) >= 3:
            ups_name = parts[2]
            if ups_name not in self._ups_data:
                return "ERR UNKNOWN-UPS"
            desc = self._ups_data[ups_name].get("ups.description", ups_name)
            return f'UPSDESC {ups_name} "{desc}"'

        if parts[1] == "NUMLOGINS" and len(parts) >= 3:
            return f"NUMLOGINS {parts[2]} 1"

        return "ERR INVALID-ARGUMENT"
```

`src/nut_up/upsd_client.py`:
```python
from __future__ import annotations

import asyncio
import re
from typing import Self


class NUTError(Exception):
    """Error returned by upsd."""

    def __init__(self, code: str, detail: str = ""):
        self.code = code
        self.detail = detail
        super().__init__(f"{code} {detail}".strip())


def _parse_quoted(s: str) -> str:
    """Extract value from a NUT quoted string like '"some value"'."""
    s = s.strip()
    if s.startswith('"') and s.endswith('"'):
        s = s[1:-1]
    return s.replace('\\"', '"').replace("\\\\", "\\")


class UpsdClient:
    """Async TCP client for the NUT upsd protocol."""

    def __init__(self, host: str = "127.0.0.1", port: int = 3493, timeout: float = 10.0):
        self.host = host
        self.port = port
        self.timeout = timeout
        self._reader: asyncio.StreamReader | None = None
        self._writer: asyncio.StreamWriter | None = None

    async def connect(self) -> None:
        self._reader, self._writer = await asyncio.wait_for(
            asyncio.open_connection(self.host, self.port),
            timeout=self.timeout,
        )

    async def close(self) -> None:
        if self._writer:
            try:
                await self._send("LOGOUT")
            except Exception:
                pass
            self._writer.close()
            self._writer = None
            self._reader = None

    async def __aenter__(self) -> Self:
        await self.connect()
        return self

    async def __aexit__(self, *exc) -> None:
        await self.close()

    async def _send(self, command: str) -> str:
        """Send a command, return the raw response line."""
        assert self._writer and self._reader
        self._writer.write((command + "\n").encode())
        await self._writer.drain()
        line = await asyncio.wait_for(self._reader.readline(), timeout=self.timeout)
        return line.decode().strip()

    async def _send_list(self, command: str) -> list[str]:
        """Send a LIST command, return all lines between BEGIN/END."""
        assert self._writer and self._reader
        self._writer.write((command + "\n").encode())
        await self._writer.drain()
        lines = []
        while True:
            line = await asyncio.wait_for(self._reader.readline(), timeout=self.timeout)
            text = line.decode().strip()
            if text.startswith("ERR "):
                code = text[4:].split()[0]
                raise NUTError(code, text[4:])
            lines.append(text)
            if text.startswith("END LIST"):
                break
        return [l for l in lines if not l.startswith("BEGIN LIST") and not l.startswith("END LIST")]

    def _check_error(self, response: str) -> str:
        if response.startswith("ERR "):
            code = response[4:].split()[0]
            raise NUTError(code, response[4:])
        return response

    async def server_version(self) -> str:
        return self._check_error(await self._send("VER"))

    async def list_ups(self) -> list[dict[str, str]]:
        lines = await self._send_list("LIST UPS")
        result = []
        for line in lines:
            m = re.match(r'UPS\s+(\S+)\s+"(.*)"', line)
            if m:
                result.append({"name": m.group(1), "description": _parse_quoted(f'"{m.group(2)}"')})

    async def list_vars(self, ups: str) -> dict[str, str]:
```

```python
from __future__ import annotations

import time
from pathlib import Path
from typing import Any

HEADER = "# Generated by nut-up -- do not edit manually\n\n"


class NUTConfigWriter:
    def __init__(self, conf_dir: str | Path = "/etc/nut"):
        self.conf_dir = Path(conf_dir)

    def _write(self, filename: str, content: str) -> Path:
        self.conf_dir.mkdir(parents=True, exist_ok=True)
        path = self.conf_dir / filename
        path.write_text(HEADER + content)
        return path

    def backup_existing(self) -> list[Path]:
        """Backup any existing config files. Returns list of backup paths."""
        backed_up = []
        ts = int(time.time())
        for name in ("nut.conf", "ups.conf", "upsd.conf", "upsd.users", "upsmon.conf"):
            path = self.conf_dir / name
            if path.exists():
                backup = path.with_suffix(f".bak.{ts}")
                backup.write_text(path.read_text())
                backed_up.append(backup)
        return backed_up

    def write_nut_conf(self, mode: str) -> Path:
        return self._write("nut.conf", f"MODE={mode}\n")

    def write_ups_conf(self, ups_list: list[dict[str, Any]]) -> Path:
        lines = []
        for ups in ups_list:
            lines.append(f'[{ups["name"]}]')
            lines.append(f'    driver = {ups["driver"]}')
            lines.append(f'    port = {ups["port"]}')
            if ups.get("desc"):
                lines.append(f'    desc = "{ups["desc"]}"')
            for k, v in ups.get("extra", {}).items():
                lines.append(f"    {k} = {v}")
            lines.append("")
        return self._write("ups.conf", "\n".join(lines) + "\n")

    def write_upsd_conf(self, listen_addresses: list[str] | None = None, port: int = 3493) -> Path:
        addrs = listen_addresses or ["127.0.0.1"]
        lines = [f"LISTEN {addr} {port}" for addr in addrs]
        lines.append("MAXAGE 15")
        return self._write("upsd.conf", "\n".join(lines) + "\n")

    def write_upsd_users(self, admin_password: str, monitor_password: str) -> Path:
        content = f"""[admin]
    password = {admin_password}
    actions = SET
    actions = FSD
    instcmds = ALL

[upsmon_primary]
    password = {monitor_password}
    upsmon primary

[upsmon_secondary]
    password = {monitor_password}
    upsmon secondary
"""
        return self._write("upsd.users", content)

    def write_upsmon_conf(self, monitors: list[dict[str, str]]) -> Path:
        lines = []
        for m in monitors:
            lines.append(
                f'MONITOR {m["ups"]}@{m["host"]} 1 {m["user"]} {m["password"]} {m["type"]}'
            )
        lines.extend([
            "",
            "MINSUPPLIES 1",
            'SHUTDOWNCMD "/sbin/shutdown -h +0"',
            "POWERDOWNFLAG /etc/killpower",
            "",
            "POLLFREQ 5",
            "POLLFREQALERT 5",
            "DEADTIME 15",
            "FINALDELAY 5",
            "",
            'NOTIFYMSG ONLINE  "UPS %s on line power"',
            'NOTIFYMSG ONBATT  "UPS %s on battery!"',
            'NOTIFYMSG LOWBATT "UPS %s battery is low!"',
            "",
            "NOTIFYFLAG ONLINE  SYSLOG",
            "NOTIFYFLAG ONBATT  SYSLOG+WALL",
            "NOTIFYFLAG LOWBATT SYSLOG+WALL",
        ])
        return self._write("upsmon.conf", "\n".join(lines) + "\n")

    def write_server(
        self,
        mode: str,
        ups_list: list[dict[str, Any]],
        admin_password: str,
        monitor_password: str,
        listen_addresses: list[str] | None = None,
    ) -> None:
        """Write all 5 config files for a NUT server."""
        if listen_addresses is None:
            listen_addresses = ["0.0.0.0"] if mode == "netserver" else ["127.0.0.1"]

        self.write_nut_conf(mode)
        self.write_ups_conf(ups_list)
        self.write_upsd_conf(listen_addresses)

            "type": "primary",
        }])


class NUTConfigReader:
    """Read existing NUT config files (for display/migration)."""

    def __init__(self, conf_dir: str | Path = "/etc/nut"):
        self.conf_dir = Path(conf_dir)

    def read_mode(self) -> str | None:
        path = self.conf_dir / "nut.conf"
        if not path.exists():
            return None
        for line in path.read_text().splitlines():
            line = line.strip()
            if line.startswith("MODE="):

        return None
```

```python
from __future__ import annotations

import re
import subprocess


def parse_nut_scanner_output(output: str) -> list[dict]:
    """Parse nut-scanner INI-style output into structured UPS definitions."""
    devices = []
    current: dict | None = None

    for line in output.splitlines():
        line = line.strip()
        if not line:
            continue
        m = re.match(r"\[(.+)\]", line)
        if m:
            if current:
                devices.append(current)
            current = {"name": m.group(1), "driver": "", "port": "auto", "desc": "", "extra": {}}
            continue
        if current and "=" in line:
            key, _, value = line.partition("=")
            key = key.strip()
            value = value.strip().strip('"')
            if key == "driver":
                current["driver"] = value
            elif key == "port":
                current["port"] = value
            elif key == "product":
                current["desc"] = value
                current["extra"][key] = value
            else:
                current["extra"][key] = value

    if current:
        devices.append(current)
    return devices


def scan_usb() -> list[dict]:
    """Run nut-scanner -U and return detected UPS devices."""
    try:
        result = subprocess.run(
            ["nut-scanner", "-U"],
            capture_output=True,
            text=True,
            timeout=30,
        )
        if result.returncode != 0:
            return []
        return parse_nut_scanner_output(result.stdout)
    except (FileNotFoundError, subprocess.TimeoutExpired):
        return []
```

```python
    if reader.has_existing_config():

    # Scan for UPS

    monitor_pw = secrets.token_urlsafe(16)
```

```python
from __future__ import annotations

from fastapi import WebSocket


class ConnectionManager:
    def __init__(self):
        self.active: list[WebSocket] = []

    async def connect(self, ws: WebSocket) -> None:
        await ws.accept()
        self.active.append(ws)

    def disconnect(self, ws: WebSocket) -> None:
        if ws in self.active:
            self.active.remove(ws)

    async def broadcast(self, data: dict) -> None:
        for ws in self.active[:]:
            try:
                await ws.send_json(data)
            except Exception:
                self.active.remove(ws)
```

```python
_ws_manager: ConnectionManager | None = None
_upsd_client: UpsdClient | None = None
_state_manager: StateManager | None = None
_latest_ups_data: dict = {}


def get_ws_manager() -> ConnectionManager:
    assert _ws_manager is not None
    return _ws_manager

    assert _upsd_client is not None
    return _upsd_client


def get_state_manager() -> StateManager:
    assert _state_manager is not None

    return _latest_ups_data


async def _poll_ups(client: UpsdClient, manager: ConnectionManager) -> None:
    """Background loop: poll upsd every 2s, broadcast to WebSocket clients."""
    global _latest_ups_data
    while True:
        try:
            ups_list = await client.list_ups()
            data = {}
            for ups in ups_list:
                name = ups["name"]
                try:
                    variables = await client.list_vars(name)
                    data[name] = {
                        "name": name,
                        "description": ups["description"],
                        "variables": variables,
                        "status": variables.get("ups.status", "unknown"),
                        "battery_charge": variables.get("battery.charge"),
                        "battery_runtime": variables.get("battery.runtime"),
                        "load": variables.get("ups.load"),
                        "input_voltage": variables.get("input.voltage"),
                        "output_voltage": variables.get("output.voltage"),
                    }
                except NUTError:
                    data[name] = {"name": name, "description": ups["description"], "error": True}
            _latest_ups_data = data
            await manager.broadcast({"type": "ups_data", "data": data})
        except (NUTError, OSError):
            _latest_ups_data = {}
            await manager.broadcast({"type": "error", "message": "Lost connection to upsd"})
            try:
                await client.close()
            except Exception:
                pass
            await asyncio.sleep(5)
            try:
                await client.connect()
            except Exception:
                pass
            continue
        await asyncio.sleep(2)


@asynccontextmanager
async def lifespan(app: FastAPI):
    global _ws_manager, _upsd_client, _state_manager

    _ws_manager = ConnectionManager()
    _state_manager = StateManager(
        Path(os.environ.get("NUT_UP_STATE_FILE", "/opt/nut-up/state.json"))
    )

    upsd_host = os.environ.get("NUT_UP_UPSD_HOST", "127.0.0.1")
    upsd_port = int(os.environ.get("NUT_UP_UPSD_PORT", "3493"))
    _upsd_client = UpsdClient(upsd_host, upsd_port)

    poll_task = None
    try:
        await _upsd_client.connect()
        poll_task = asyncio.create_task(_poll_ups(_upsd_client, _ws_manager))
    except OSError:
        pass  # upsd not available yet

    yield

    if poll_task:
        poll_task.cancel()
        try:
            await poll_task
        except asyncio.CancelledError:
            pass
    if _upsd_client:
        await _upsd_client.close()


def create_app() -> FastAPI:
    app = FastAPI(title="nut-up", version="0.1.0", lifespan=lifespan, docs_url="/api/docs")

    from nut_up.routers import api, commands, enrollment, ws
    app.include_router(api.router, prefix="/api", tags=["ups"])
    app.include_router(commands.router, prefix="/api", tags=["commands"])
    app.include_router(enrollment.router, tags=["enrollment"])
    app.include_router(ws.router)

    # Static files -- mount LAST so API routes take priority
    if STATIC_DIR.exists():
        app.mount("/", StaticFiles(directory=STATIC_DIR, html=True), name="static")

    return app
```

```python
from __future__ import annotations

import secrets
from datetime import datetime, timezone, timedelta

from nut_up.state import StateManager


class EnrollmentManager:
    def __init__(self, state: StateManager):
        self.state = state

    def create_key(self, label: str, hours: int = 24) -> str:
        token = "nutup_" + secrets.token_urlsafe(16)
        now = datetime.now(timezone.utc)
        expires = now + timedelta(hours=hours)

        def add(data):
            data["keys"][token] = {
                "label": label,
                "created_at": now.isoformat(),
                "expires_at": expires.isoformat(),
            }
            return data

        self.state.update(add)
        return token

    def validate_key(self, key: str) -> bool:
        data = self.state.load()
        entry = data["keys"].get(key)
        if not entry:
            return False
        expires = datetime.fromisoformat(entry["expires_at"])
        return datetime.now(timezone.utc) < expires

    def revoke_key(self, key: str) -> None:
        def remove(data):
            data["keys"].pop(key, None)
            return data
        self.state.update(remove)

    def list_keys(self) -> list[dict]:
        data = self.state.load()
        result = []
        now = datetime.now(timezone.utc)
        for token, info in data["keys"].items():
            expires = datetime.fromisoformat(info["expires_at"])
            result.append({
                "key": token,
                "label": info["label"],
                "created_at": info["created_at"],
                "expires_at": info["expires_at"],
                "expired": now >= expires,
            })
        return result

    def record_enrollment(self, key: str, hostname: str, ip: str) -> None:
        now = datetime.now(timezone.utc)
        def add(data):
            data["clients"][hostname] = {
                "ip": ip,
                "enrolled_at": now.isoformat(),
                "key_used": key,
            }
            return data
        self.state.update(add)

    def list_clients(self) -> list[dict]:
        data = self.state.load()
        return [
            {"hostname": h, **info}
            for h, info in data["clients"].items()
        ]

    def generate_enrollment_script(

# Server: {server_host}:{server_port}

echo "=== nut-up client enrollment ==="

# Install NUT client
if ! command -v upsmon &>/dev/null; then
    echo "Installing nut-client..."
    apt-get update -qq && apt-get install -y -qq nut-client
fi

# Configure NUT client
echo "Configuring NUT client..."

cat > /etc/nut/nut.conf << 'NUTCONF'
MODE=netclient
NUTCONF

cat > /etc/nut/upsmon.conf << 'UPSMONCONF'
MONITOR {ups_name}@{server_host} 1 upsmon_secondary {monitor_password} secondary
MINSUPPLIES 1
SHUTDOWNCMD "/sbin/shutdown -h +0"
POWERDOWNFLAG /etc/killpower
POLLFREQ 5
POLLFREQALERT 5
DEADTIME 15
FINALDELAY 5
NOTIFYMSG ONLINE  "UPS %s on line power"
NOTIFYMSG ONBATT  "UPS %s on battery!"
NOTIFYMSG LOWBATT "UPS %s battery is low!"
NOTIFYFLAG ONLINE  SYSLOG
NOTIFYFLAG ONBATT  SYSLOG+WALL
NOTIFYFLAG LOWBATT SYSLOG+WALL
```

`src/nut_up/routers/enrollment.py`:
```python
from __future__ import annotations

import socket

from fastapi import APIRouter, HTTPException, Request
from fastapi.responses import PlainTextResponse
from pydantic import BaseModel

from nut_up.app import get_state_manager
from nut_up.enrollment import EnrollmentManager

router = APIRouter()


class CreateKeyRequest(BaseModel):
    label: str
    hours: int = 24


class EnrollConfirmRequest(BaseModel):
    key: str
    hostname: str
    ip: str


def _em() -> EnrollmentManager:
    return EnrollmentManager(get_state_manager())


def _server_config() -> dict:
    state = get_state_manager()
    data = state.load()
    return data.get("server", {})


@router.post("/api/keys")
async def create_key(req: CreateKeyRequest):
    em = _em()
    key = em.create_key(label=req.label, hours=req.hours)
    return {"key": key}


@router.get("/api/keys")
async def list_keys():
    return _em().list_keys()


@router.delete("/api/keys/{key}")
async def delete_key(key: str):
    _em().revoke_key(key)
    return {"status": "ok"}


@router.get("/api/clients")
async def list_clients():
    return _em().list_clients()


@router.get("/setup")
async def setup_endpoint(key: str, request: Request):
    """Enrollment endpoint: curl http://server:3494/setup?key=... | sudo bash"""
    em = _em()
    if not em.validate_key(key):
        raise HTTPException(403, "Invalid or expired enrollment key")

    server = _server_config()
    server_host = request.headers.get("host", "").split(":")[0]
    if not server_host or server_host in ("localhost", "127.0.0.1"):
        server_host = _get_local_ip()

    script = em.generate_enrollment_script(
        key=key,
        server_host=server_host,
        server_port=request.url.port or 3494,
        ups_name=server.get("ups_name", "ups"),
        monitor_password=server.get("monitor_password", ""),
    )
    return PlainTextResponse(script, media_type="text/x-shellscript")


@router.post("/api/enroll/confirm")
async def enroll_confirm(req: EnrollConfirmRequest):
    em = _em()
    if not em.validate_key(req.key):
        raise HTTPException(403, "Invalid or expired enrollment key")
    em.record_enrollment(req.key, hostname=req.hostname, ip=req.ip)
    return {"status": "ok", "hostname": req.hostname}


@router.get("/api/manual-setup")
async def manual_setup():
    """Connection details for manual client setup (OPNsense, Synology, etc.)."""
    server = _server_config()
    return {
        "ups_name": server.get("ups_name", "ups"),
        "server_host": _get_local_ip(),
        "server_port": 3493,
        "monitor_user": "upsmon_secondary",
        "monitor_password": server.get("monitor_password", ""),
    }


def _get_local_ip() -> str:
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        s.connect(("8.8.8.8", 80))
        ip = s.getsockname()[0]
        s.close()
        return ip
    except Exception:
        return "127.0.0.1"
```

```javascript
// In app.js -- include this helper
escapeHtml(str) {
```
